async function generateImage() { const prompt = $("prompt").value.trim(); if (!prompt) { alert("اكتب وصف الصورة أولاً!"); return; } $("emptyState").classList.add("hidden"); $("resultState").classList.add("hidden"); $("errorState").classList.add("hidden"); $("loadingState").classList.remove("hidden"); $("generateBtn").disabled = true; const width = Number($("width").value); const ratio = Number($("ratio").value); const height = Math.round(width / ratio); const fullPrompt = `${prompt}, ${selectedStyle}, masterpiece, highly detailed, professional composition`; const seed = Math.floor(Math.random() * 9999999); const imageUrl = "https://image.pollinations.ai/prompt/" + encodeURIComponent(fullPrompt) + `?width=${width}&height=${height}&seed=${seed}&nologo=true`; const img = new Image(); img.onload = function () { currentImage = imageUrl; $("resultImage").src = imageUrl; $("resultState").classList.remove("hidden"); $("loadingState").classList.add("hidden"); $("generateBtn").disabled = false; gallery.unshift({ url: imageUrl, prompt: prompt }); renderGallery(); }; img.onerror = function () { $("loadingState").classList.add("hidden"); $("errorState").classList.remove("hidden"); $("errorMessage").textContent = "تعذر توليد الصورة. قد تكون الخدمة القديمة متوقفة أو غير متاحة حالياً."; $("generateBtn").disabled = false; }; img.src = imageUrl; }
